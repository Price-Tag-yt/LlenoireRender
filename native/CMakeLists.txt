cmake_minimum_required(VERSION 3.22.1)
project(LlenoireRender LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(LlenoireRender SHARED core/renderer.cpp config/config.cpp diagnostics/log.cpp diagnostics/caps.cpp state_cache/state_cache.cpp shader/preprocessor.cpp texture/texture_cache.cpp framebuffer/framebuffer.cpp sync/frame_pacer.cpp gles_backend/gles_backend.cpp egl/egl_forwarder.cpp android/jni.cpp)
target_include_directories(LlenoireRender PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_library(log-lib log)
find_library(dl-lib dl)
find_library(egl-lib EGL)
find_library(gles-lib GLESv2)
target_link_libraries(LlenoireRender ${log-lib} ${dl-lib} ${egl-lib} ${gles-lib})
target_compile_definitions(LlenoireRender PRIVATE LLR_VERSION="0.1.0" LLR_TARGET_MC="1.12.2")
if(CMAKE_BUILD_TYPE STREQUAL "Release")
 target_compile_options(LlenoireRender PRIVATE -O3 -ffunction-sections -fdata-sections)
 target_link_options(LlenoireRender PRIVATE -Wl,--gc-sections)
endif()
